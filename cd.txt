pipeline {
    agent any

    stages {
        stage('git clone') {
            steps {
                git branch: 'main', url: 'https://github.com/immanoj22/jenkins.git'
            }
        }

        stage('deploy in eks cluster') {
            steps {
                sh 'kubectl apply -f deployment.yaml'
            }
        }
    }   // <-- stages closes here

    post {
        success {
            emailext(
                from: 's.manojraj193@gmail.com',
                to: 's.manojraj193@gmail.com',
                subject: "SUCCESS: ${env.JOB_NAME} #${env.BUILD_NUMBER}",
                body: """Build succeeded.

Job: ${env.JOB_NAME}
Build number: ${env.BUILD_NUMBER}
Details: ${env.BUILD_URL}"""
            )
        }
        failure {
            emailext(
                from: 's.manojraj193@gmail.com',
                to: 's.manojraj193@gmail.com',
                subject: "FAILED: ${env.JOB_NAME} #${env.BUILD_NUMBER}",
                body: """Build failed.

Job: ${env.JOB_NAME}
Build number: ${env.BUILD_NUMBER}
Check the console output: ${env.BUILD_URL}console"""
            )
        }
    }
}
